    <script>
        // État du calculateur
        let binCount = 0;

        // Mettre à jour le nombre de bacs (minimum 0, maximum 10)
        function changeBinCount(delta) {
            binCount = Math.max(0, Math.min(10, binCount + delta));
            document.getElementById('bin-count-display').textContent = binCount;
            document.getElementById('bin-subtotal').textContent = (binCount * 20) + ' $';
            updateEstimate();
        }

        // Calculer et afficher le total estimé
        function updateEstimate() {
            let total = 0;
            const hasLeaves = document.getElementById('calc-leaves').checked;
            const hasSnow = document.getElementById('calc-snow').checked;
            const hasLawn = document.getElementById('calc-lawn').checked;

            if (hasLeaves) total += 25;
            total += binCount * 20;
            if (hasSnow) total += 20;
            if (hasLawn) total += 50;

            document.getElementById('total-price').textContent = total + ' $';

            const variableNote = document.getElementById('variable-note');
            if (hasLawn) {
                variableNote.classList.remove('hidden');
            } else {
                variableNote.classList.add('hidden');
            }
        }

        // Ajouter un service directement depuis les cartes de tarifs
        function selectServiceFromCard(serviceType) {
            if (serviceType === 'leaves') {
                document.getElementById('calc-leaves').checked = true;
            } else if (serviceType === 'bins') {
                changeBinCount(1);
            } else if (serviceType === 'snow') {
                document.getElementById('calc-snow').checked = true;
            } else if (serviceType === 'lawn') {
                document.getElementById('calc-lawn').checked = true;
            }
            updateEstimate();

            // Défilement fluide vers le calculateur
            document.getElementById('calculateur').scrollIntoView({ behavior: 'smooth' });
        }

        // Transférer les choix du calculateur vers le formulaire de contact
        // Gestion de la soumission du formulaire avec confirmation visuelle et envoi courriel pré-rempli
        function handleFormSubmit(event) {
            event.preventDefault();

            const name = document.getElementById('client-name').value.trim();
            const phone = document.getElementById('client-phone').value.trim();
            const address = document.getElementById('client-address').value.trim();
            const details = document.getElementById('client-message').value.trim();

            // Récupérer les services cochés
            const checkedServices = Array.from(document.querySelectorAll('input[name="services"]:checked'))
                .map(cb => `- ${cb.value}`);

            // Construire le sujet et le corps du courriel vers prosduquartier.equipe@gmail.com
            const subject = encodeURIComponent(`Demande de service - ${name} (${address})`);
            const bodyLines = [
                `Bonjour Jacob, Zakary et Noah,`,
                ``,
                `Voici ma demande de service :`,
                `• Nom : ${name}`,
                `• Téléphone / Courriel : ${phone}`,
                `• Adresse : ${address}`,
                ``,
                `Services demandés :`,
                checkedServices.length > 0 ? checkedServices.join('\n') : `- À préciser`,
                ``,
                details ? `Détails supplémentaires :\n${details}` : ``
            ];
            const body = encodeURIComponent(bodyLines.join('\n'));

            // Ouvrir l'application courriel du client avec le message prêt à envoyer
            window.location.href = `mailto:prosduquartier.equipe@gmail.com?subject=${subject}&body=${body}`;

            const confirmationDetails = document.getElementById('confirmation-details');
            confirmationDetails.innerHTML = `Merci <strong>${escapeHTML(name)}</strong> ! Votre demande pour le <strong>${escapeHTML(address)}</strong> est prête à être envoyée à <strong>prosduquartier.equipe@gmail.com</strong>. Jacob, Zakary ou Noah vous contactera très rapidement.`;

            document.getElementById('booking-form').classList.add('hidden');
            document.getElementById('confirmation-box').classList.remove('hidden');
        }

        // Réinitialiser le formulaire
        function resetBookingForm() {
            document.getElementById('booking-form').reset();
            document.getElementById('confirmation-box').classList.add('hidden');
            document.getElementById('booking-form').classList.remove('hidden');
        }

        // Utilitaire pour sécuriser l'affichage du texte saisi
        function escapeHTML(str) {
            const div = document.createElement('div');
            div.textContent = str;
            return div.innerHTML;
        }
    </script>
